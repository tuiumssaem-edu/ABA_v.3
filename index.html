구글 드라이브 연동 웹앱 URL

Google Apps Script로 배포한 웹앱 URL을 입력하시면, 앱 상단의 [구글 시트 저장] 클릭 시 선생님 구글 드라이브에 스프레드시트가 자동 생성되며 학생별 탭과 헤더가 자동으로 기록됩니다.

구글 앱스 스크립트(GAS) 설정 3단계 안내
  1. script.google.com에 접속하여 [새 프로젝트]를 만듭니다.
  2. 기존 코드를 모두 지우고, 아래의 [코드 복사] 버튼을 눌러 복사한 스크립트를 붙여넣고 저장합니다.
  3. 우측 상단 [배포] → [새 배포] 선택 후, 톱니바퀴에서 웹앱(Web app)을 선택합니다.
    • 다음 사용자 계정으로 실행: 나(내 계정)
    • 액세스 권한이 있는 사용자: 모든 사용자(Anyone)
  4. 생성된 웹앱 URL을 복사하여 위의 입력창에 붙여넣고 [저장]을 누릅니다.
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var now = new Date();
    var dateString = Utilities.formatDate(now, "Asia/Seoul", "yyyy-MM-dd_HHmm");
    
    // 1. 구글 드라이브에 새 스프레드시트 생성
    var ss = SpreadsheetApp.create("[행동기록부] " + dateString);
    var defaultSheet = ss.getActiveSheet();
    
    // 2. 학생별 개별 시트(탭) 생성 및 데이터 기입
    data.students.forEach(function(student, index) {
      var sheet = ss.insertSheet(student.name);
      
      // A. 헤더 스타일 지정
      var headerStyle = SpreadsheetApp.newTextStyle().setBold(true).build();
      
      // 1) ABC 관찰 기록 헤더
      sheet.appendRow(["[1. ABC 사건 관찰 기록]"]);
      sheet.appendRow(["일자", "시각", "장소/교시", "선행사건(A)", "행동(B)", "후속결과(C)", "추정목적(기능)"]);
      var abcRange = sheet.getRange(sheet.getLastRow(), 1, 1, 7);
      abcRange.setBackground("#e2e8f0").setTextStyle(headerStyle);
      
      if (student.abcRecords && student.abcRecords.length > 0) {
        student.abcRecords.forEach(function(r) {
          sheet.appendRow([r.date, r.time, r.context, r.antecedent, r.behavior, r.consequence, r.hypothesis]);
        });
      } else {
        sheet.appendRow(["(기록 없음)"]);
      }
      
      sheet.appendRow([]); // 빈 줄
      
      // 2) 빈도 및 강도 기록 헤더
      sheet.appendRow(["[2. 빈도 및 3단계 강도 기록]"]);
      sheet.appendRow(["일자", "시각", "표적행동명", "강도수준(1~3)", "강도명칭"]);
      var freqRange = sheet.getRange(sheet.getLastRow(), 1, 1, 5);
      freqRange.setBackground("#e2e8f0").setTextStyle(headerStyle);
      
      if (student.freqRecords && student.freqRecords.length > 0) {
        student.freqRecords.forEach(function(r) {
          sheet.appendRow([r.date, r.time, r.behaviorName, r.intensity, r.levelName]);
        });
      } else {
        sheet.appendRow(["(기록 없음)"]);
      }
      
      sheet.appendRow([]); // 빈 줄
      
      // 3) 지속시간(스톱워치) 계측 헤더
      sheet.appendRow(["[3. 표적행동별 지속시간 기록]"]);
      sheet.appendRow(["일자", "시작시각", "종료시각", "표적행동명", "지속시간(초)", "환산시간"]);
      var durRange = sheet.getRange(sheet.getLastRow(), 1, 1, 6);
      durRange.setBackground("#e2e8f0").setTextStyle(headerStyle);
      
      if (student.durationRecords && student.durationRecords.length > 0) {
        student.durationRecords.forEach(function(r) {
          sheet.appendRow([r.date, r.startTime, r.endTime, r.behaviorName, r.durationSeconds, r.durationFormatted]);
        });
      } else {
        sheet.appendRow(["(기록 없음)"]);
      }
      
      // 열 너비 자동 맞춤
      for (var col = 1; col <= 7; col++) {
        sheet.autoResizeColumn(col);
      }
    });
    
    // 기본 생성된 빈 시트 제거
    try { ss.deleteSheet(defaultSheet); } catch(err) {}
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      url: ss.getUrl()
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
